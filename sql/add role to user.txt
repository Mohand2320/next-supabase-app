-- ════════════════════════════════════════════════════════════
-- 0. VÉRIFICATIONS UTILES (à utiliser à tout moment)
-- ════════════════════════════════════════════════════════════

-- 0.1 Voir tous les utilisateurs inscrits (auth.users)
SELECT id, email, created_at 
FROM auth.users 
ORDER BY created_at DESC;

-- 0.2 Voir tous les profils configurés (user_profiles)
SELECT 
  up.user_id,
  au.email,
  up.role,
  up.is_active,
  up.dentiste_id,
  up.assistant_id,
  up.created_at
FROM user_profiles up
JOIN auth.users au ON au.id = up.user_id
ORDER BY up.created_at DESC;

-- 0.3 Voir tous les dentistes
SELECT id, nom, prenom, specialite, numero_rpps FROM dentistes;

-- 0.4 Voir tous les assistants
SELECT id, nom, prenom, login FROM assistants;

-- 0.5 Trouver l'ID d'un utilisateur par son email
SELECT id, email FROM auth.users WHERE email = 'amoura@gmail.com';

-- ════════════════════════════════════════════════════════════
-- 5. DONNER LE RÔLE ADMIN À UN DENTISTE EXISTANT

-- ⚠ ATTENTION : Changer le rôle en 'admin' supprime l'accès
--   aux fonctions de dentiste (séances, suppression patients, etc.)
-- ⚠ SOLUTION RECOMMANDÉE : Utiliser is_admin (voir section 6)

-- Si vous voulez quand même le faire :
UPDATE user_profiles
SET role = 'admin', dentiste_id = NULL, assistant_id = NULL
WHERE user_id = 'ID_USER_ICI';

-- ════════════════════════════════════════════════════════════
-- 9. CHANGER LE RÔLE D'UN UTILISATEUR
-- ════════════════════════════════════════════════════════════

-- 9.1 Passer un assistant → dentiste
-- Étape A : Créer l'entrée dentiste
INSERT INTO dentistes (id, nom, prenom) VALUES (gen_random_uuid(), 'NOM', 'PRENOM')
RETURNING id;

-- Étape B : Mettre à jour le profil
UPDATE user_profiles
SET role = 'dentiste', dentiste_id = 'ID_DENTISTE_ICI', assistant_id = NULL
WHERE user_id = 'ID_USER_ICI';

-- 9.2 Passer un dentiste → assistant
INSERT INTO assistants (id, nom, prenom, login) VALUES (gen_random_uuid(), 'NOM', 'PRENOM', 'login')
RETURNING id;

UPDATE user_profiles
SET role = 'assistant', assistant_id = 'ID_ASSISTANT_ICI', dentiste_id = NULL
WHERE user_id = 'ID_USER_ICI';
